### EXEMPLO SIMPLES DE VALIDAÇÃO

def soma(a: int, b: int) -> int:

return a + b

print(soma(10,20)) #

print(soma(“10”,“20”)) #

class Numeros(BaseModel):

a: int

b: int

def soma(numeros: Numeros) -> int:

return numeros.a + numeros.b

numeros = Numeros(a="10", b="20")

print(soma(numeros))
